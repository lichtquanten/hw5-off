cmake_minimum_required(VERSION 3.10)
project(CS104-HW5-Test-Suite LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED TRUE)
set(CMAKE_CXX_EXTENSIONS FALSE)
set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-g3 -O0")
add_compile_options(-Wall)

if(NOT DEFINED HW_DIR)
    get_filename_component(HW_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    set(HW_DIR "${HW_DIR}" CACHE PATH "Path to the assignment directory")
endif()
get_filename_component(HW_DIR "${HW_DIR}" REALPATH)
if(NOT IS_DIRECTORY "${HW_DIR}/src")
    message(FATAL_ERROR "Cannot find src in ${HW_DIR}. Set -DHW_DIR=<assignment directory>.")
endif()

find_package(GTest REQUIRED)
find_package(Threads REQUIRED)
find_program(VALGRIND NAMES valgrind)
if(NOT VALGRIND)
    message(FATAL_ERROR "Valgrind is required to check memory errors and peak memory usage.")
endif()
set(VALGRIND_ARGS --tool=memcheck --leak-check=yes --trace-children=no)

include(test_cmake_lib/CustomTests.cmake)
enable_testing()
add_subdirectory(shipwreck_tests)
add_custom_target(grade DEPENDS check)
